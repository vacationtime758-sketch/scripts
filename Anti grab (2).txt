local savedCFrame = nil

local function getRoot()
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    return char:FindFirstChild("HumanoidRootPart")
end

local function updateUI()
    if savedCFrame then
        StatusLabel.Text = "Estado: Punto Guardado ✓"
        StatusLabel.TextColor3 = Color3.fromRGB(80, 220, 80)
        FloatTpBtn.Text = "⚡"
        FloatTpBtn.BackgroundColor3 = Color3.fromRGB(40, 180, 40)
    else
        StatusLabel.Text = "Estado: Sin Marcar"
        StatusLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
        FloatTpBtn.Text = "📌"
        FloatTpBtn.BackgroundColor3 = Color3.fromRGB(240, 180, 0)
    end
end

local function processTpAction()
    local root = getRoot()
    if not root then return end

    if not savedCFrame then
        savedCFrame = root.CFrame
    else
        root.CFrame = savedCFrame
    end
    updateUI()
end

FloatTpBtn.MouseButton1Click:Connect(processTpAction)
MenuTpBtn.MouseButton1Click:Connect(processTpAction)

ResetBtn.MouseButton1Click:Connect(function()
    savedCFrame = nil
    updateUI()
end)

CloseBtn.MouseButton1Click:Connect(function()
    MainFrame.Visible = false
    OpenMenuBtn.Visible = true
end)

OpenMenuBtn.MouseButton1Click:Connect(function()
    MainFrame.Visible = true
    OpenMenuBtn.Visible = false
end)

-- Lógica del Cañón
local function tpAndActivateCannon()
    local hrp = getRoot()
    if not hrp then return end

    local cannonTarget = nil
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local name = obj.Name:lower()
            if (name:find("cannon") or name:find("cañon")) and not name:find("fito") then
                cannonTarget = obj
                break
            end
        end
    end

    if cannonTarget then
        local targetPart = cannonTarget:IsA("Model") and cannonTarget.PrimaryPart or cannonTarget
        if not targetPart and cannonTarget:IsA("Model") then
            for _, p in ipairs(cannonTarget:GetDescendants()) do
                if p:IsA("BasePart") then
                    targetPart = p
                    break
                end
            end
        end
        
        if targetPart then
            hrp.CFrame = targetPart.CFrame + Vector3.new(0, 3, 0)
            task.wait(0.05)
            
            local prompt = cannonTarget:FindFirstChildWhichIsA("ProximityPrompt", true)
            if prompt then
                fireproximityprompt(prompt)
            else
                local click = cannonTarget:FindFirstChildWhichIsA("ClickDetector", true)
                if click then
                    fireclickdetector(click)
                else
                    firetouchinterest(hrp, targetPart, 0)
                    task.wait(0.02)
                    firetouchinterest(hrp, targetPart, 1)
                end
            end
        end
    end
end

CannonBtn.MouseButton1Click:Connect(tpAndActivateCannon)
