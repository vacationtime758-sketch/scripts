local miniDragging, miniMoved = false, false
local miniDragStart, miniStartPos = nil, nil

local function bindMiniDrag(handle)
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            miniDragging, miniMoved = true, false
            miniDragStart, miniStartPos = i.Position, miniWrap.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then miniDragging = false end
            end)
        end
    end)
end

    if miniDragging and (i.UserInputType == Enum.UserInputType.MouseMovement
    or i.UserInputType == Enum.UserInputType.Touch) then

        if math.abs(d.X) > 6 or math.abs(d.Y) > 6 then miniMoved = true end
        miniWrap.Position = UDim2.new(miniStartPos.X.Scale, miniStartPos.X.Offset + d.X,
                                      miniStartPos.Y.Scale, miniStartPos.Y.Offset + d.Y)

end)

--===================== LOGIC =====================
local function getRoot()
    local char = lp.Character
    if not char then char = lp.CharacterAdded:Wait() end
    return char:WaitForChild("HumanoidRootPart", 3)
end

local function resolvePart(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then return obj end
    if obj.PrimaryPart then return obj.PrimaryPart end
    for _, p in ipairs(obj:GetDescendants()) do
        if p:IsA("BasePart") then return p end
    end
    return nil
end

local function findCannons()
    local found = {}
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local n = obj.Name:lower()
            if (n:find("cannon") or n:find("cañon")) and not n:find("fito") then
                local dup = false
                for _, ex in ipairs(found) do
                    if obj:IsDescendantOf(ex) or ex:IsDescendantOf(obj) then
                        dup = true
                        break
                    end
                end
                if not dup then table.insert(found, obj) end
            end
        end
    end

    if hrp then
        table.sort(found, function(a, b)
            local pa, pb = resolvePart(a), resolvePart(b)
            if not pa or not pb then return false end
            return (pa.Position - hrp.Position).Magnitude < (pb.Position - hrp.Position).Magnitude
        end)
    end
    return found
end

local function activate(obj)
    if not obj or not obj.Parent then return false end

    local roots = { obj }
    local up = obj.Parent
    for _ = 1, 2 do
        if up and up ~= Workspace then
            table.insert(roots, up)
            up = up.Parent
        end
    end

    for _, root in ipairs(roots) do
        local prompt = root:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt and fireproximityprompt then
            pcall(fireproximityprompt, prompt)
            return true
        end
        local click = root:FindFirstChildWhichIsA("ClickDetector", true)
        if click and fireclickdetector then
            pcall(fireclickdetector, click)
            return true
        end
    end

    local part = resolvePart(obj)
    if part and firetouchinterest then
        local hrp = getRoot()
        if hrp then
            pcall(firetouchinterest, hrp, part, 0)
            task.wait(0.05)
            pcall(firetouchinterest, hrp, part, 1)
            return true
        end
    end
    return false
end

local function flashCircle(ok)
    local col = ok and GREEN or ORANGE
    Circle.BackgroundColor3 = col
    cs.Color = col
    task.delay(0.35, function()
        if not Circle.Parent then return end
        Tween:Create(Circle, TweenInfo.new(0.25), { BackgroundColor3 = Color3.fromRGB(0, 0, 0) }):Play()
        Tween:Create(cs, TweenInfo.new(0.25), { Color = ORANGE }):Play()
    end)
end

local function goTo(obj)
    if teleporting then return end
    obj = obj or selected
    if not obj or not obj.Parent then
        local cannons = findCannons()
        obj = cannons[1]
        if not obj then
            status.Text = "No cannon found"
            status.TextColor3 = Color3.fromRGB(255, 120, 120)
            flashCircle(false)
            return
        end
    end

    local part = resolvePart(obj)
    local hrp = getRoot()
    if not part or not hrp then
        status.Text = "Cannon has no part"
        status.TextColor3 = Color3.fromRGB(255, 120, 120)
        flashCircle(false)
        return
    end

    teleporting = true
    selected = obj
    status.Text = "Going to " .. obj.Name
    status.TextColor3 = ORANGE

    hrp.CFrame = part.CFrame + Vector3.new(0, 4, 0)
    hrp.AssemblyLinearVelocity = Vector3.zero
    task.wait(0.12)

    local ok = activate(obj)
    flashCircle(ok)
    status.Text = ok and ("Activated " .. obj.Name) or ("At " .. obj.Name .. " (no prompt)")
    status.TextColor3 = ok and Color3.fromRGB(120, 240, 130) or Color3.fromRGB(255, 190, 90)
    task.wait(0.15)
    teleporting = false
end

local function renderList()
    for _, c in ipairs(list:GetChildren()) do
        if c:IsA("TextButton") or c:IsA("Frame") then c:Destroy() end
    end

    local cannons = findCannons()
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")

    if #cannons == 0 then
        status.Text = "No cannons found"
        status.TextColor3 = Color3.fromRGB(255, 120, 120)
    else
        status.Text = string.format("%d cannon(s) found", #cannons)
        status.TextColor3 = Color3.fromRGB(185, 185, 195)
    end

    if selected and not selected.Parent then selected = nil end

    for i, obj in ipairs(cannons) do
        local part = resolvePart(obj)
        local dist = ""
        if part and hrp then
            dist = string.format("%dm", math.floor((part.Position - hrp.Position).Magnitude))
        end

        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, -8, 0, 52)
        row.Position = UDim2.new(0, 4, 0, (i - 1) * 58)
        row.BackgroundColor3 = (selected == obj) and Color3.fromRGB(66, 40, 12) or Color3.fromRGB(24, 24, 30)
        row.BorderSizePixel = 0
        row.Parent = list
        Instance.new("UICorner", row).CornerRadius = UDim.new(0, 9)

        local info = Instance.new("TextButton")
        info.Size = UDim2.new(1, -96, 1, 0)
        info.BackgroundTransparency = 1
        info.Text = string.format("%d. %s   %s", i, obj.Name, dist)
        info.TextColor3 = (selected == obj) and ORANGE or Color3.fromRGB(225, 225, 235)
        info.Font = Enum.Font.GothamBold
        info.TextSize = 13
        info.TextXAlignment = Enum.TextXAlignment.Left
        info.Parent = row
        info.MouseButton1Click:Connect(function()
            selected = obj
            status.Text = "Selected: " .. obj.Name
            status.TextColor3 = ORANGE
            renderList()
        end)

        local goB = Instance.new("TextButton")
        goB.Size = UDim2.new(0, 82, 1, -12)
        goB.Position = UDim2.new(1, -88, 0, 6)
        goB.BackgroundColor3 = Color3.fromRGB(45, 95, 45)
        goB.Text = "GO"
        goB.TextColor3 = Color3.fromRGB(255, 255, 255)
        goB.Font = Enum.Font.GothamBlack
        goB.TextSize = 14
        goB.Parent = row
        Instance.new("UICorner", goB).CornerRadius = UDim.new(0, 8)
        goB.MouseButton1Click:Connect(function() goTo(obj) end)
    end

    list.CanvasSize = UDim2.new(0, 0, 0, math.max(0, #cannons * 58))
end

--===================== INPUT =====================
UIS.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    if rebinding then
        if input.KeyCode == Enum.KeyCode.Escape then
            bindKey = nil
            bindBtn.Text = "BIND: NONE"
        else
            bindKey = input.KeyCode
            bindBtn.Text = "BIND: " .. input.KeyCode.Name
        end
        rebinding = false
        return
    end

    if bindKey and input.KeyCode == bindKey then
        goTo(selected)
    end
end)

    end
end)
