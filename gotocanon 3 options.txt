--===================== DRAG =====================
local function draggable(obj, handle)
    local dragging, startPos, dragStart, moved = false, nil, nil, false
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            dragStart, startPos = i.Position, obj.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
        or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            if math.abs(d.X) > 4 or math.abs(d.Y) > 4 then moved = true end
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                     startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    return function() return moved end
end
draggable(Frame, TitleBar)

--===================== LOGIC =====================
local function getRoot()
    local char = lp.Character
    if not char then char = lp.CharacterAdded:Wait() end
    return char:WaitForChild("HumanoidRootPart", 3)
end

local function resolvePart(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then return obj end
    if obj.PrimaryPart then return obj.PrimaryPart end
    for _, p in ipairs(obj:GetDescendants()) do
        if p:IsA("BasePart") then return p end
    end
    return nil
end

local function findCannons()
    local found = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local n = obj.Name:lower()
            if (n:find("cannon") or n:find("cañon")) and not n:find("fito") then
                local dup = false
                for _, ex in ipairs(found) do
                    if obj:IsDescendantOf(ex) or ex:IsDescendantOf(obj) then
                        dup = true
                        break
                    end
                end
                if not dup then table.insert(found, obj) end
            end
        end
    end
    return found
end

local function activate(obj)
    if not obj or not obj.Parent then return false end

    -- search the object, then walk up a couple of ancestors
    local searchRoots = { obj }
    local up = obj.Parent
    for _ = 1, 2 do
        if up and up ~= Workspace then
            table.insert(searchRoots, up)
            up = up.Parent
        end
    end

    for _, root in ipairs(searchRoots) do
        local prompt = root:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt and fireproximityprompt then
            pcall(fireproximityprompt, prompt)
            return true
        end
        local click = root:FindFirstChildWhichIsA("ClickDetector", true)
        if click and fireclickdetector then
            pcall(fireclickdetector, click)
            return true
        end
    end

    local part = resolvePart(obj)
    if part and firetouchinterest then
        local hrp = getRoot()
        if hrp then
            pcall(firetouchinterest, hrp, part, 0)
            task.wait(0.05)
            pcall(firetouchinterest, hrp, part, 1)
            return true
        end
    end
    return false
end

local function goToSelected()
    if teleporting then return end
    if not selected or not selected.Parent then
        status.Text = "No cannon selected"
        status.TextColor3 = Color3.fromRGB(255, 120, 120)
        return
    end

    local part = resolvePart(selected)
    local hrp = getRoot()
    if not part or not hrp then
        status.Text = "Cannon has no part"
        status.TextColor3 = Color3.fromRGB(255, 120, 120)
        return
    end

    teleporting = true
    status.Text = "Going to " .. selected.Name
    status.TextColor3 = ORANGE

    hrp.CFrame = part.CFrame + Vector3.new(0, 4, 0)
    task.spawn(function()
        local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
        if hum and hrp then hrp.AssemblyLinearVelocity = Vector3.zero end
    end)
    task.wait(0.12)

    local ok = activate(selected)
    status.Text = ok and ("Activated " .. selected.Name) or ("At " .. selected.Name .. " (no prompt found)")
    status.TextColor3 = ok and Color3.fromRGB(120, 240, 130) or Color3.fromRGB(255, 190, 90)
    task.wait(0.15)
    teleporting = false
end

local function renderList()
    for _, c in ipairs(list:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    rows = {}

    local cannons = findCannons()
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")

    if #cannons == 0 then
        status.Text = "No cannons found in this map"
        status.TextColor3 = Color3.fromRGB(255, 120, 120)
    else
        status.Text = string.format("%d cannon(s) found", #cannons)
        status.TextColor3 = Color3.fromRGB(180, 180, 190)
    end

    if selected and not selected.Parent then selected = nil end

    for i, obj in ipairs(cannons) do
        local part = resolvePart(obj)
        local dist = ""
        if part and hrp then
            dist = string.format("  [%d]", math.floor((part.Position - hrp.Position).Magnitude))
        end

        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, -6, 0, 28)
        b.Position = UDim2.new(0, 3, 0, (i - 1) * 30)
        b.BackgroundColor3 = (selected == obj) and Color3.fromRGB(70, 42, 12) or Color3.fromRGB(24, 24, 30)
        b.TextColor3 = (selected == obj) and ORANGE or Color3.fromRGB(220, 220, 230)
        b.Font = Enum.Font.GothamBold
        b.TextSize = 11
        b.TextXAlignment = Enum.TextXAlignment.Left
        b.Text = string.format("%d. %s%s", i, obj.Name, dist)
        b.Parent = list
        Instance.new("UICorner", b).CornerRadius = UDim.new(0, 6)

        b.MouseButton1Click:Connect(function()
            selected = obj
            status.Text = "Selected: " .. obj.Name
            status.TextColor3 = ORANGE
            renderList()
        end)

        rows[obj] = b
    end

    list.CanvasSize = UDim2.new(0, 0, 0, math.max(0, #cannons * 30))
end

--===================== INPUT =====================
UIS.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    -- rebind swallows the key so it never fires the action
    if rebinding then
        if input.KeyCode == Enum.KeyCode.Escape then
            bindKey = nil
            bindBtn.Text = "BIND: NONE"
        else
            bindKey = input.KeyCode
            bindBtn.Text = "BIND: " .. input.KeyCode.Name
        end
        bindBtn.TextColor3 = ORANGE
        rebinding = false
        return
    end

    if bindKey and input.KeyCode == bindKey then
        goToSelected()
    elseif input.KeyCode == Enum.KeyCode.Insert then
        if Frame.Visible then
            Frame.Visible = false
            Circle.Visible = true
        else
            Circle.Visible = false
            Frame.Visible = true
        end
    end
end)

--===================== AUTO REFRESH =====================

    refreshTimer += dt
    if refreshTimer >= 3 then
        refreshTimer = 0
        if Frame.Visible then renderList() end
    end
end)
