local function log(s)
    lines[#lines + 1] = tostring(s)
    while #lines > 25 do table.remove(lines, 1) end
    if box and box.Parent then box.Text = table.concat(lines, "\n") end
    print(s)
end

local function mk(text, y, fn)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1, -20, 0, 26)
    b.Position = UDim2.fromOffset(10, y)
    b.BackgroundColor3 = Color3.fromRGB(34, 34, 44)
    b.BorderSizePixel = 0
    b.Font = Enum.Font.GothamBold
    b.TextSize = 9
    b.TextColor3 = Color3.fromRGB(215, 215, 228)
    b.Text = text
    b.Parent = main
    cr(b, 5)
    b.MouseButton1Click:Connect(fn)
end

mk("1. VIM SendKeyDown/Up", 24, function()
    pcall(function()
        VIM:SendKeyDown(Enum.KeyCode.F, true)
        task.delay(0.4, function() pcall(function() VIM:SendKeyUp(Enum.KeyCode.F) end) end)
    end)
    log("1 fired")
end)

mk("2. VIM SendKeyEvent", 54, function()
    pcall(function()
        VIM:SendKeyEvent(true, Enum.KeyCode.F, false, game)
        task.delay(0.4, function() pcall(function() VIM:SendKeyEvent(false, Enum.KeyCode.F, false, game) end) end)
    end)
    log("2 fired")
end)

mk("3. keypress", 84, function()
    if keypress then
        pcall(keypress, "f")
        task.delay(0.4, function() if keyrelease then pcall(keyrelease, "f") end end)
        log("3 fired")
    else
        log("3 unavailable")
    end
end)

mk("4. VIM keyboard 4-arg", 114, function()

        task.delay(0.4, function() pcall(function() VIM:SendKeyUp(Enum.KeyCode.F, false, game) end) end)
    end)
    log("4 fired")
end)

task.spawn(function()
    local ch = lp.Character or lp.CharacterAdded:Wait()
    hum = ch:WaitForChild("Humanoid", 8)
    anim = hum and hum:FindFirstChildOfClass("Animator")
    if anim then
        anim.AnimationPlayed:Connect(function(t)
            local n = (t.Animation and t.Animation.Name) or "?"
            log("GAME PLAYED: " .. n)
        end)
    end
    local ok, re = pcall(function()
        return game:GetService("ReplicatedStorage").Packages.Knit.Services.CombatService:WaitForChild("RE", 15)
    end)
    if ok and re then
        local ok2, d = pcall(function() return re:WaitForChild("TookDamage", 10) end)
        if ok2 and d then
            d.OnClientEvent:Connect(function(v)
                if typeof(v) == "number" then log(("%.1f%s"):format(v, v <= 2 and "  << HALVED" or "")) end
            end)
        end
    end
    log("ready")
end)
