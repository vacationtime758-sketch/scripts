local dragMoved = {}
local function makeDraggable(obj)
    local dragging, dragStart, startPos, moved = false, nil, nil, false
    dragMoved[obj] = function() return moved end
    obj.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            dragStart, startPos = input.Position, obj.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            if math.abs(d.X) > 4 or math.abs(d.Y) > 4 then moved = true end
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                     startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function valToT(v, mn, mx, exp)
    if exp then return math.log(v / mn) / math.log(mx / mn) end
    return (v - mn) / (mx - mn)
end
local function tToVal(t, mn, mx, exp)
    if exp then return mn * (mx / mn) ^ t end
    return mn + t * (mx - mn)
end

--===================== LOGIC =====================
local function weldedPlayer()
    local char = lp.Character
    if not char then return nil end
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") or d:IsA("Snap") then
            local a, b = d.Part0, d.Part1
            if a and b then
                local other = nil
                if a:IsDescendantOf(char) and not b:IsDescendantOf(char) then other = b
                elseif b:IsDescendantOf(char) and not a:IsDescendantOf(char) then other = a end
                if other then
                    local model = other:FindFirstAncestorOfClass("Model")
                    local plr = model and Players:GetPlayerFromCharacter(model)
                    if plr and plr ~= lp then return plr end
                end
            end
        end
    end
    return nil
end

local function grabbedPlayer()
    local w = weldedPlayer()
    if w then return w end
    local char = lp.Character
    if os.clock() - lastGrabTime < 0.35 and lastGrabTarget and lastGrabTarget.Character then
        local myHRP = char and char:FindFirstChild("HumanoidRootPart")
        local tHRP  = lastGrabTarget.Character:FindFirstChild("HumanoidRootPart")
        if myHRP and tHRP and (myHRP.Position - tHRP.Position).Magnitude < 12 then
            return lastGrabTarget
        end
    end
    return nil
end

local function targetPlayer()
    if CFG.exactName then
        local p = Players:FindFirstChild(CFG.exactName)
        if p and p ~= lp and not isExcluded(p)
        and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            return p
        end
    end

    local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")

    if #STRAFE_TARGETS > 0 then
        local best, bd = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= lp and not isExcluded(p) and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local nm = p.Name:lower()
                    local dn = (p.DisplayName or ""):lower()
                    local match = false
                    for _, s in ipairs(STRAFE_TARGETS) do
                        local q = s:lower()
                        if nm:find(q, 1, true) or dn:find(q, 1, true) then match = true; break end
                    end
                    if match then
                        if root then
                            local d = (hrp.Position - root.Position).Magnitude
                            if d < bd then bd = d; best = p end
                        else
                            return p
                        end
                    end
                end
            end
        end
        if best then return best end
    end

    if not root then return nil end
    local best, bd = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and not isExcluded(p) and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - root.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    return best
end

local function buildCF()
    local t = targetPlayer()
    if not t or not t.Character then return nil end
    local head = t.Character:FindFirstChild("Head") or t.Character:FindFirstChild("HumanoidRootPart")
    local hrp  = t.Character:FindFirstChild("HumanoidRootPart")
    if not head or not hrp then return nil end

    local yaw
    if CFG.yawMode == 1 then yaw = math.rad(hrp.Orientation.Y)
    elseif CFG.yawMode == 2 then yaw = freeYaw
    else yaw = math.rad(spinAcc) end

    local base = CFrame.Angles(0, yaw, 0)
    local pos  = head.Position
        + Vector3.new(0, CFG.height, 0)
        + base.RightVector * CFG.side
        + (-base.LookVector) * CFG.back
        + moveOffset

    return CFrame.new(pos) * base * CFrame.Angles(math.rad(CFG.pitch), 0, math.rad(CFG.roll))
end

local function strafeEnable()
    local char = lp.Character
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return false end
    freeYaw = math.rad(hrp.Orientation.Y)
    saved = { anchored = hrp.Anchored, platform = hum.PlatformStand, autorotate = hum.AutoRotate }
    hum.PlatformStand = true
    hum.AutoRotate = false
    if not CFG.sync then hrp.Anchored = true end
    pcall(function() for _, tr in ipairs(hum:GetPlayingAnimationTracks()) do tr:Stop() end end)
    moveOffset = Vector3.zero
    frozenCF = nil
    strafeGrabCF = nil
    return true
end

local function strafeDisable()
    local char = lp.Character
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if saved and hum and hrp then
        hrp.Anchored      = saved.anchored
        hum.PlatformStand = saved.platform
        hum.AutoRotate    = saved.autorotate
    end
    saved, frozenCF, strafeGrabCF = nil, nil, nil
    moveOffset = Vector3.zero
end

local function strafeToggle()
    if CFG.strafeOn then CFG.strafeOn = false; strafeDisable()
    elseif strafeEnable() then CFG.strafeOn = true end
end

local function wasdDir()
    local cam = workspace.CurrentCamera
    if not cam then return Vector3.zero end
    local dir = Vector3.zero
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cam.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cam.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
    return dir
end

--===================== MAIN LOOP =====================
RunService.RenderStepped:Connect(function(dt)
    local char = lp.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp then return end

    if CFG.yawMode == 3 then spinAcc += CFG.spin * dt end

    local welded = weldedPlayer()

    if CFG.antiFling then
        local grabbed = welded or grabbedPlayer()
        if grabbed then
            if not grabActive then
                grabActive = true
                grabDampUntil = os.clock() + CFG.grabDamp
            end
            if os.clock() < grabDampUntil then
                hrp.AssemblyLinearVelocity  = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            else
                local v = hrp.AssemblyLinearVelocity
                if v.Magnitude > CFG.maxGrabSpeed then
                    hrp.AssemblyLinearVelocity = v.Unit * CFG.maxGrabSpeed
                end
                hrp.AssemblyAngularVelocity = Vector3.zero
            end
        else
            grabActive = false
        end
    else
        grabActive = false
    end

    if not CFG.strafeOn then strafeGrabCF = nil; return end

    -- STABILIZE: while welded to the grabbed player, stop re-reading their
    -- live position (that's what caused the run-away). Hold our own CF.
    if welded and CFG.stabOnGrab then
        if not strafeGrabCF then strafeGrabCF = hrp.CFrame end
        if CFG.move then
            local dir = wasdDir()
            if dir.Magnitude > 0.01 then
                strafeGrabCF = CFrame.new(strafeGrabCF.Position + dir.Unit * CFG.strafe * dt)
                    * strafeGrabCF.Rotation
            end
        end
        hrp.CFrame = strafeGrabCF
        if CFG.sync then
            hrp.AssemblyLinearVelocity  = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
            if hum then hum.MoveDirection = Vector3.zero end
        end
        return
    end
    strafeGrabCF = nil

    if CFG.move then
        local dir = wasdDir()
        if dir.Magnitude > 0.01 then moveOffset += dir.Unit * CFG.strafe * dt end
    end

    local cf
    if CFG.hold and frozenCF then cf = frozenCF
    else
        cf = buildCF()
        if not cf then return end
        if CFG.hold then frozenCF = cf end
    end

    hrp.CFrame = cf
    if CFG.sync then
        hrp.AssemblyLinearVelocity  = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        if hum then hum.MoveDirection = Vector3.zero end
    end
end)

    if CFG.strafeOn then CFG.strafeOn = false; strafeToggle() end
end)

local function pickGrabTarget()
    local myChar = lp.Character
    local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if #GRAB_TARGETS == 0 then
        if not myHRP then return nil end
        local closest, cd = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= lp and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local d = (hrp.Position - myHRP.Position).Magnitude
                    if d < cd then cd = d; closest = p end
                end
            end
        end
        return closest
    end
    local search = GRAB_TARGETS[1]:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and (p.Name:lower():find(search, 1, true)
        or (p.DisplayName and p.DisplayName:lower():find(search, 1, true))) then
            return p
        end
    end
    return nil
end

local function doGrab()
    local myHRP = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end
    if not GrabRF then
        pcall(function() GrabRF = ReplicatedStorage.Packages.Knit.Services.CombatService.RF.Grab end)
    end
    if not GrabRF then return end
    lastGrabTarget = pickGrabTarget()
    lastGrabTime = os.clock()
    pcall(function() GrabRF:InvokeServer() end)
end

local function doTeleport()
    if not TP_MODE then return end
    local myHRP = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end
    local t = pickGrabTarget()
    if not t or not t.Character then return end
    local tHRP = t.Character:FindFirstChild("HumanoidRootPart")
    if tHRP then myHRP.CFrame = tHRP.CFrame end
end

local function startSpam()
    task.spawn(function()
        while spamActive do
            doGrab()
            if SPAM_DELAY > 0 then task.wait(SPAM_DELAY) end
        end
    end)
end

local function updateESP()
    local myRoot = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    for player, obj in pairs(espObjects) do
        local char = player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")

        local valid = ESP.enabled and hrp ~= nil and hum ~= nil and hum.Health > 0
        if valid and not ESP.showExcluded and isExcluded(player) then valid = false end
        if valid and ESP.teamCheck and player.Team ~= nil and player.Team == lp.Team then valid = false end

        local d = 0
        if valid and myRoot and hrp then
            d = (hrp.Position - myRoot.Position).Magnitude
            if ESP.maxDist > 0 and d > ESP.maxDist then valid = false end
        end

        obj.highlight.Adornee = valid and char or nil
        obj.box.Adornee = valid and char or nil
        obj.billboard.Adornee = valid and head or nil

        if valid then
            obj.highlight.Enabled = ESP.chams
            obj.box.Visible = ESP.box
            obj.billboard.Enabled = ESP.name or ESP.dist or ESP.health
            obj.name.Visible = ESP.name
            obj.hpBg.Visible = ESP.health
            obj.dist.Visible = ESP.dist
            obj.name.Text = player.Name
            obj.billboard.MaxDistance = ESP.maxDist
            if ESP.health and hum then
                obj.hpFill.Size = UDim2.new(math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1), 0, 1, 0)
            end
            if ESP.dist then
                obj.dist.Text = string.format("%d studs", math.floor(d))
            end
        else
            obj.highlight.Enabled = false

--===================== KEYBINDS =====================
UserInputService.InputBegan:Connect(function(input, gp)
    if gp or bindingFor then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if input.KeyCode == GRAB_KEYBIND then
        spamActive = not spamActive
        refreshGrabUI()
        if spamActive then startSpam() end
    elseif input.KeyCode == TP_KEYBIND then
        doTeleport()
    elseif input.KeyCode == ESP_KEYBIND then
        ESP.enabled = not ESP.enabled
    elseif input.KeyCode == Enum.KeyCode.Insert then
        main.Visible = not main.Visible
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if not bindingFor then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if input.KeyCode == Enum.KeyCode.Escape then
        if bindingFor == "grab" then GRAB_KEYBIND = nil; grabBindBtn.Text = "Grab: None"
        elseif bindingFor == "esp" then ESP_KEYBIND = nil; eBindBtn.Text = "Key: None"
        else TP_KEYBIND = nil; tpBindBtn.Text = "TP: None" end
    else
        if bindingFor == "grab" then GRAB_KEYBIND = input.KeyCode; grabBindBtn.Text = "Grab: " .. input.KeyCode.Name
        elseif bindingFor == "esp" then ESP_KEYBIND = input.KeyCode; eBindBtn.Text = "Key: " .. input.KeyCode.Name
        else TP_KEYBIND = input.KeyCode; tpBindBtn.Text = "TP: " .. input.KeyCode.Name end
    end
    bindingFor = nil
end)
