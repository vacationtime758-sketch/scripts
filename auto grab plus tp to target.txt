--==================================================================
--  AUTO GRAB
--  G = toggle spam grab | TP bind = teleport (only when TP mode ON)
--==================================================================

local Players          = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local ReplicatedStorage= game:GetService("ReplicatedStorage")
local CoreGui          = game:GetService("CoreGui")
local LocalPlayer      = Players.LocalPlayer

--===================== CONFIG =====================
local GRAB_KEYBIND     = Enum.KeyCode.G   -- toggles spam grab (rebindable)
local TP_KEYBIND       = Enum.KeyCode.T   -- teleport to target, only when TP mode ON (rebindable)
local TP_MODE_DEFAULT  = false            -- TP mode starts off
local SPAM_DELAY       = 0.01             -- lower = faster spam
local TARGETS          = {}               -- {"name1","name2"} empty = nearest player
--==================================================

-- ---------- remote ----------
local GrabRF = nil
task.spawn(function()
    task.wait(1)
    pcall(function()
        GrabRF = ReplicatedStorage.Packages.Knit.Services.CombatService.RF.Grab
    end)
end)

local grabTargetIndex = 1
local teleportEnabled = TP_MODE_DEFAULT
local spamActive      = false
local bindingFor      = nil   -- "grab" | "tp" while rebinding

-- ---------- target picker ----------
local function pickTarget()
    local myChar = LocalPlayer.Character
    local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")

    if #TARGETS == 0 then
        if not myHRP then return nil end
        local closest, closestDist = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local d = (hrp.Position - myHRP.Position).Magnitude
                    if d < closestDist then closestDist = d; closest = p end
                end
            end
        end
        return closest
    else
        if grabTargetIndex > #TARGETS then grabTargetIndex = 1 end
        local search = TARGETS[grabTargetIndex]:lower()
        grabTargetIndex = grabTargetIndex + 1
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and
               (p.Name:lower():find(search, 1, true) or
               (p.DisplayName and p.DisplayName:lower():find(search, 1, true))) then
                return p
            end
        end
        return nil
    end
end

-- ---------- teleport to target (only when TP mode is on) ----------
local function doTeleport()
    if not teleportEnabled then return end
    local myChar = LocalPlayer.Character
    local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end

    local targetPlayer = pickTarget()
    if not targetPlayer or not targetPlayer.Character then return end
    local targetHRP = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not targetHRP then return end

    myHRP.CFrame = targetHRP.CFrame
end

-- ---------- grab (in place, no teleport) ----------
local function doGrab()
    local myChar = LocalPlayer.Character
    local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end
    if not GrabRF then
        pcall(function()
            GrabRF = ReplicatedStorage.Packages.Knit.Services.CombatService.RF.Grab
        end)
    end
    if not GrabRF then return end

    pcall(function() GrabRF:InvokeServer() end)
end

-- ---------- spam loop ----------
local function startSpam()
    task.spawn(function()
        while spamActive do
            doGrab()
            if SPAM_DELAY > 0 then task.wait(SPAM_DELAY) end
        end
    end)
end

-- drag
local dragging, dragStart, startPos
frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = frame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local function refreshList()
    for _, c in ipairs(list:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    for i, name in ipairs(TARGETS) do
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, -6, 0, 22)
        b.Position = UDim2.new(0, 3, 0, (i - 1) * 24)
        b.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
        b.Text = name .. "   x"
        b.TextColor3 = Color3.fromRGB(200, 200, 220)
        b.Font = Enum.Font.Gotham
        b.TextSize = 12
        b.Parent = list
        Instance.new("UICorner", b).CornerRadius = UDim.new(0, 4)
        b.MouseButton1Click:Connect(function()
            table.remove(TARGETS, i)
            refreshList()
        end)
    end
    list.CanvasSize = UDim2.new(0, 0, 0, #TARGETS * 24)
end

local function addTarget()
    local txt = input.Text:gsub("%s", "")
    if txt ~= "" then
        table.insert(TARGETS, txt)
        input.Text = ""
        refreshList()
    end
end

addBtn.MouseButton1Click:Connect(addTarget)
input.FocusLost:Connect(function() addTarget() end)

local function updateStatus()
    teleportBtn.Text = "TP Mode: " .. (teleportEnabled and "ON" or "OFF")
    teleportBtn.TextColor3 = teleportEnabled and Color3.fromRGB(120, 255, 150) or Color3.fromRGB(255, 140, 140)

    status.Text = "AUTO GRAB: " .. (spamActive and "ON" or "OFF")
    status.TextColor3 = spamActive and Color3.fromRGB(120, 255, 150) or Color3.fromRGB(255, 140, 140)
    status.BackgroundColor3 = spamActive and Color3.fromRGB(25, 45, 30) or Color3.fromRGB(25, 25, 35)
end

refreshList()
updateStatus()

--==================================================================
--  KEYBINDS
--==================================================================
-- G -> toggle spam grab (always)
UserInputService.InputBegan:Connect(function(input, gp)
    if gp or bindingFor then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if input.KeyCode ~= GRAB_KEYBIND then return end

    spamActive = not spamActive
    updateStatus()
    if spamActive then startSpam() end
end)

-- TP bind -> teleport, only when TP mode is ON
UserInputService.InputBegan:Connect(function(input, gp)
    if gp or bindingFor then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if input.KeyCode == TP_KEYBIND then
        doTeleport()
    end
end)

-- rebind handler
UserInputService.InputBegan:Connect(function(input, gp)
    if not bindingFor then return end
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    if input.KeyCode == Enum.KeyCode.Escape then
        if bindingFor == "grab" then
            GRAB_KEYBIND = nil
            bindBtn.Text = "Grab: None"
        else
            TP_KEYBIND = nil
            tpBindBtn.Text = "TP: None"
        end
    else
        if bindingFor == "grab" then
            GRAB_KEYBIND = input.KeyCode
            bindBtn.Text = "Grab: " .. input.KeyCode.Name
        else
            TP_KEYBIND = input.KeyCode
            tpBindBtn.Text = "TP: " .. input.KeyCode.Name
        end
    end
    bindingFor = nil
end)
