-- Parse avoid list
local function updateAvoidList()
    local text = AvoidBox.Text
    avoidList = {}
    for name in text:gmatch("[^,]+") do
        name = name:gsub("^%s+", ""):gsub("%s+$", "")
        if name ~= "" then
            table.insert(avoidList, name:lower())
        end
    end
    AvoidStatus.Text = "Avoiding: " .. #avoidList .. " player" .. (#avoidList == 1 and "" or "s")
end

AvoidBox.FocusLost:Connect(updateAvoidList)

-- Check if player should be avoided
local function shouldAvoid(player)
    local playerName = player.Name:lower()
    local displayName = player.DisplayName:lower()
    for _, avoidName in ipairs(avoidList) do
        if playerName:find(avoidName) or displayName:find(avoidName) then
            return true
        end
    end
    return false
end

-- Functions
local function updateStatus(text, color)
    StatusLabel.Text = text
    StatusLabel.TextColor3 = color or Color3.fromRGB(200, 200, 200)
end

local function getKeyCodeFromString(str)
    str = str:gsub("%s+", "")
    for _, key in pairs(Enum.KeyCode:GetEnumItems()) do
        if key.Name:lower() == str:lower() then
            return key
        end
    end
    return nil
end

-- Get distance (cached for performance)
local function getDistance(player)
    if not playerCache[player] then playerCache[player] = {} end
    local now = tick()
    if playerCache[player].distTime and now - playerCache[player].distTime < 0.1 then
        return playerCache[player].dist
    end
    
    local myChar = LocalPlayer.Character
    local theirChar = player.Character
    if not myChar or not theirChar then 
        playerCache[player].dist = math.huge
        playerCache[player].distTime = now
        return math.huge 
    end
    
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    local theirRoot = theirChar:FindFirstChild("HumanoidRootPart")
    if not myRoot or not theirRoot then 
        playerCache[player].dist = math.huge
        playerCache[player].distTime = now
        return math.huge 
    end
    
    local dist = (myRoot.Position - theirRoot.Position).Magnitude
    playerCache[player].dist = dist
    playerCache[player].distTime = now
    return dist
end

-- REAL-TIME wall check for triggerbot (NO CACHE)
local function isPlayerVisibleRealTime(player)
    if not player.Character then return false end
    
    local myPos = Camera.CFrame.Position
    local character = player.Character
    local head = character:FindFirstChild("Head")
    
    if not head then return false end
    
    -- Raycast from camera to head
    local direction = (head.Position - myPos)
    local distance = direction.Magnitude
    direction = direction.Unit
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {LocalPlayer.Character, character}
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    raycastParams.IgnoreWater = true
    
    local result = workspace:Raycast(myPos, direction * distance, raycastParams)
    
    -- If no hit, player is visible
    if not result then return true end
    
    -- If hit something transparent, still visible
    if result.Instance.Transparency > 0.9 then return true end
    
    -- Blocked by wall
    return false
end

-- Cached wall check for ESP only
local function isPlayerVisibleCached(player)
    local now = tick()
    if visibilityCache[player] and now - visibilityCache[player].time < 0.2 then
        return visibilityCache[player].visible
    end
    
    local visible = isPlayerVisibleRealTime(player)
    visibilityCache[player] = {time = now, visible = visible}
    return visible
end

-- Check if valid target
local function isValidTarget(target)
    if not target or not target.Parent then return false end
    local character = target.Parent
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local plr = Players:FindFirstChild(character.Name)
    
    if not humanoid or humanoid.Health <= 0 then return false end
    if not plr then return false end
    if shouldAvoid(plr) then return false end
    if getDistance(plr) > maxDistance then return false end
    return true
end

-- Check crosshair on target with REAL-TIME wall check for triggerbot
local function isCrosshairOnTarget()
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if shouldAvoid(player) then continue end
        
        if player.Character then
            local dist = getDistance(player)
            if dist > maxDistance then continue end
            
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            local head = player.Character:FindFirstChild("Head")
            
            if not humanoid or humanoid.Health <= 0 then continue end
            
            if head then
                local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local screenDist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if screenDist < 20 then
                        -- REAL-TIME wall check - NO CACHE for triggerbot
                        if wallCheckEnabled and not isPlayerVisibleRealTime(player) then
                            return false, nil, nil -- Wall blocked, don't shoot
                        end
                        return true, head, player
                    end
                end
            end
        end
    end
    return false, nil, nil
end

-- Get closest target
local function getClosestTarget()
    local closest = nil
    local shortestDist = math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if shouldAvoid(player) then continue end
        
        if player.Character then
            local dist = getDistance(player)
            if dist > maxDistance then continue end
            
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            local head = player.Character:FindFirstChild("Head")
            
            if not humanoid or humanoid.Health <= 0 then continue end
            
            if head then
                local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local screenDist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if screenDist < detectionRadius and screenDist < shortestDist then
                        if wallCheckEnabled and not isPlayerVisibleCached(player) then continue end
                        shortestDist = screenDist
                        closest = head
                    end
                end
            end
        end
    end
    return closest
end

-- Main loop (single RenderStepped)
local frameCount = 0
RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    
    -- ESP update (every 3rd frame)
    if espEnabled and frameCount % 3 == 0 then
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            
            local esp = getESP(player)
            local char = player.Character
            if char then
                local head = char:FindFirstChild("Head")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if head and humanoid and humanoid.Health > 0 then
                    local dist = getDistance(player)
                    local avoided = shouldAvoid(player)
                    local visible = isPlayerVisibleCached(player)
                    
                    esp.frame.Adornee = head
                    esp.frame.Enabled = dist <= maxDistance
                    esp.dist.Text = math.floor(dist) .. "m"
                    
                    if avoided then
                        esp.name.TextColor3 = Color3.fromRGB(100, 100, 100)
                        esp.status.Text = "[AVOIDED]"
                        esp.status.TextColor3 = Color3.fromRGB(150, 150, 150)
                    elseif visible then
                        esp.name.TextColor3 = Color3.fromRGB(255, 50, 50)
                        esp.status.Text = "[TARGET]"
                        esp.status.TextColor3 = Color3.fromRGB(0, 255, 0)
                    else
                        esp.name.TextColor3 = Color3.fromRGB(255, 150, 50)
                        esp.status.Text = "[WALL]"
                        esp.status.TextColor3 = Color3.fromRGB(255, 100, 100)
                    end
                else
                    esp.frame.Enabled = false
                end
            else
                esp.frame.Enabled = false
            end
        end
    end
    
    -- Softaim (every frame for smoothness)
    if softaimEnabled and (not onlyInMatch or isMatchActive) then
        if not currentTarget or not isValidTarget(currentTarget) then
            currentTarget = getClosestTarget()
        end
        
        if currentTarget and currentTarget.Parent then
            local targetPos = currentTarget.Position
            local root = currentTarget.Parent:FindFirstChild("HumanoidRootPart")
            if root then
                targetPos = targetPos + (root.Velocity * 0.03)
            end
            local targetCFrame = CFrame.new(Camera.CFrame.Position, targetPos)
            Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, aimbotStrength)
        end
    end
    
    -- Triggerbot (every 2nd frame) with REAL-TIME wall check
    if triggerbotEnabled and (not onlyInMatch or isMatchActive) and frameCount % 2 == 0 then
        local onTarget, targetHead, targetPlayer = isCrosshairOnTarget()
        
        if onTarget and targetPlayer then
            -- Double-check visibility in real-time before shooting
            if not wallCheckEnabled or isPlayerVisibleRealTime(targetPlayer) then
                local now = tick()
                if now - lastShot >= triggerDelay then
                    pcall(function()
                        mouse1click()
                    end)
                    lastShot = now
                    isOnTarget = true
                end
            else
                isOnTarget = false
            end
        else
            isOnTarget = false
        end
    end
end)

    visibilityCache[player] = nil
    if espObjects[player] then
        espObjects[player].frame:Destroy()

-- Slider functions (properly cleaned up)
local function setupSlider(slider, fill, callback)
    local connection = nil
    
    local function update(input)
        local sliderPos = slider.AbsolutePosition.X
        local sliderSize = slider.AbsoluteSize.X
        local mouseX = input.Position.X
        local relativeX = math.clamp((mouseX - sliderPos) / sliderSize, 0, 1)
        fill.Size = UDim2.new(relativeX, 0, 1, 0)
        callback(relativeX)
    end
    
    slider.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            update(input)
            
            if connection then connection:Disconnect() end
            connection = RunService.RenderStepped:Connect(function()
                if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                    local pos = UserInputService:GetMouseLocation()
                    update({Position = Vector2.new(pos.X, pos.Y)})
                else
                    connection:Disconnect()
                    connection = nil
                end
            end)
        end
    end)
end
